WG pages are always excluded — they have their own nav system. */}} {{- $isWG := $.Params.wgID -}} {{- if not $isWG -}} {{- range $.Ancestors -}} {{- if and (not $isWG) .Params.wgID -}} {{- $isWG = true -}} {{- end -}} {{- end -}} {{- end -}} {{/* Wrap everything so no output is ever produced unless all conditions pass */}} {{/* Opt-in: only render when sectionNav is explicitly true */}} {{- if and (not $isWG) (eq $.Page.Params.sectionNav true) -}} {{/* ── Find nav root and sub-sections ───────────────────────────────── */}} {{/* The root is the top of the `sectionNav` cascade — the event page — not the reader's current section. Anchored at the current section, opening an event's Sponsors page swapped the event's tabs for the sponsors section's own children, which put its "Thank you" landing page in the menu and dropped Overview, Agenda and the rest (#112). */}} {{- $navRoot := $.Page.CurrentSection -}} {{- range seq 8 -}} {{- with $navRoot.Parent -}} {{- if and (ne .Kind "home") (eq .Params.sectionNav true) -}} {{- $navRoot = . -}} {{- end -}} {{- end -}} {{- end -}} {{- $subPages := slice -}} {{- range $navRoot.Sections -}} {{- if and (ne .Params.robots "noindex") (ne .Params.noindex true) -}} {{- $subPages = $subPages | append . -}} {{- end -}} {{- end -}} {{/* If on a leaf section (e.g. agenda/), walk up to find siblings */}} {{- if eq (len $subPages) 0 -}} {{- $parent := $navRoot.Parent -}} {{- if and $parent (ne $parent.Kind "home") -}} {{- $parentSubs := slice -}} {{- range $parent.Sections -}} {{- if and (ne .Params.robots "noindex") (ne .Params.noindex true) -}} {{- $parentSubs = $parentSubs | append . -}} {{- end -}} {{- end -}} {{- if gt (len $parentSubs) 0 -}} {{- $navRoot = $parent -}} {{- $subPages = $parentSubs -}} {{- end -}} {{- end -}} {{- end -}} {{- if gt (len $subPages) 0 -}} {{- $current := $.Page.RelPermalink -}} {{- $rootHref := $navRoot.RelPermalink -}}
{{ $navRoot.LinkTitle }} {{- range $subPages }} {{ .LinkTitle }} {{- end }}
