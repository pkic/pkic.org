{{/* pagination — the list pages' page navigation. Hugo's `_internal/pagination.html` writes Bootstrap 5 markup (`pagination`, `page-item`, `page-link`). With the framework gone those classes style nothing, so the blog's pager rendered as a bulleted list of blue numbers. This writes the same DOM as `ui/Pager.tsx` instead, so one stylesheet dresses the portal's pager and the public list pages alike. A page that cannot be navigated to is a `span`, not a link with the href removed: there is nothing to activate, and `aria-disabled` says so. Expects a paginator in `.Paginator` (call `.Paginate` before including it). */}} {{- $p := .Paginator -}} {{- if gt $p.TotalPages 1 -}} {{- $window := 2 -}}
  1. {{- if $p.HasPrev }} ‹ {{- else }} ‹ {{- end }}
    2. {{- $first := 1 -}} {{- $last := $p.TotalPages -}} {{- range $p.Pagers -}} {{- $n := .PageNumber -}} {{- $near := and (ge $n (sub $p.PageNumber $window)) (le $n (add $p.PageNumber $window)) -}} {{- if or (eq $n $first) (eq $n $last) $near }}
  2. {{- if eq $n $p.PageNumber }} {{ $n }} {{- else }} {{ $n }} {{- end }}
    3. {{- else if or (eq $n (add $first 1)) (eq $n (sub $last 1)) }}
  3. …
    4. {{- end -}} {{- end }}
  4. {{- if $p.HasNext }} › {{- else }} › {{- end }}
{{- end -}}
