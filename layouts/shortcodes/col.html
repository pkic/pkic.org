{{/* col.html — responsive content-grid column for use inside {{% row %}}. Use with the percent-sign syntax so inner Markdown is rendered: {{% col size="7" %}} ## My heading Some **markdown** content here. {{% /col %}} Parameters (all optional): size — large-screen column span 1–12 (default: 12) sm — small-screen column span 1–12 (default: 12) md — medium-screen column span 1–12 (default: 12) sticky — If "true", keeps the column visible on large screens class — Extra CSS classes to add to the column div */}} {{- $size := .Get "size" | default "" -}} {{- $sm := .Get "sm" | default "" -}} {{- $md := .Get "md" | default "" -}} {{- $sticky := .Get "sticky" | default "" -}} {{- $class := .Get "class" | default "" -}} {{- $colClass := "content-col" -}} {{- with $sm }}{{- $colClass = printf "%s content-col--sm-%s" $colClass . -}}{{ end -}} {{- with $md }}{{- $colClass = printf "%s content-col--md-%s" $colClass . -}}{{ end -}} {{- with $size }}{{- $colClass = printf "%s content-col--lg-%s" $colClass . -}}{{ end -}} {{- if eq $sticky "true" }}{{- $colClass = printf "%s content-col--sticky" $colClass -}}{{ end -}} {{- with $class }}{{- $colClass = printf "%s %s" $colClass . -}}{{ end -}}
